# Tokenizer tests: exact token-id parity with llama.cpp's llama-tokenize + round-trip + streaming detokenizer.
if(NOT DEFINED IIAN_SOURCE_ROOT)
    set(IIAN_SOURCE_ROOT ${CMAKE_SOURCE_DIR})
endif()

add_executable(test-tokenizer test-tokenizer.cpp)
target_link_libraries(test-tokenizer PRIVATE iian_tokenizer ggml)
target_compile_features(test-tokenizer PRIVATE cxx_std_20)
target_compile_definitions(test-tokenizer PRIVATE
    IIAN_TOKENIZER_CORPUS="${CMAKE_CURRENT_SOURCE_DIR}/corpus.txt")
if(NOT MSVC)
    target_compile_options(test-tokenizer PRIVATE -Wall -Wextra)
endif()

# One test per model. Expected ids are generated by gen_expected.sh (llama.cpp's llama-tokenize).
set(IIAN_TOKENIZER_TEST_MODELS
    SmolLM2-135M-Instruct-F16
    Qwen2.5-0.5B-Instruct-Q4_K_M
)
foreach(model ${IIAN_TOKENIZER_TEST_MODELS})
    add_test(NAME test-tokenizer-${model}
        COMMAND test-tokenizer
            ${IIAN_SOURCE_ROOT}/models/${model}.gguf
            ${CMAKE_CURRENT_SOURCE_DIR}/expected/${model}.parse.txt
            ${CMAKE_CURRENT_SOURCE_DIR}/expected/${model}.noparse.txt)
endforeach()
